Practical 11.4

Requirement: Write a PHP function to perform following string operations:

  1. 11.4.1: print your name.
  2. 11.4.2: print the size of a string. Pass string as an argument.
  3. 11.4.3: concat two strings.
  4. 11.4.4: convert case of string.
  5. 11.4.5: find one string from another.

<?php
/**
 * Practical 11.4: PHP String Functions Suite
 * Group 10 — Government MCA College, Maninagar (GMCA)
 */

// 11.4.1 print your name
function printStudentName($name) {
    return "Student Name: " . htmlspecialchars($name);
}

// 11.4.2 print the size of a string (passed as argument)
function getStringLength($str) {
    return strlen($str);
}

// 11.4.3 concat two strings
function concatTwoStrings($str1, $str2, $delimiter = " ") {
    return $str1 . $delimiter . $str2;
}

// 11.4.4 convert case of string
function convertStringCases($str) {
    return [
        'upper' => strtoupper($str),
        'lower' => strtolower($str),
        'title' => ucwords(strtolower($str))
    ];
}

// 11.4.5 find one string from another
function searchSubstring($haystack, $needle) {
    $pos = strpos($haystack, $needle);
    if ($pos !== false) {
        return "Found at index: $pos";
    }
    return "Not Found";
}
?>
