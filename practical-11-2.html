Practical 11.2

Requirement: Write a PHP program to display current date and time and display Good Morning / Good Afternoon / Good Evening message according to current time. Uses date("H") to inspect the 24-hour hour index and triggers dynamic greetings using conditional statements (if-elseif-else).

<?php
/**
 * Practical 11.2: Display current date and time and dynamic greeting
 * Group 10 — Government MCA College, Maninagar (GMCA)
 */

date_default_timezone_set('Asia/Kolkata');

$currentDate = date("l, d F Y");
$currentTime = date("h:i:s A");
$hour = (int)date("H"); // 24-hour hour integer (0 to 23)

echo "Current Date: " . $currentDate . "<br>\n";
echo "Current Time: " . $currentTime . "<br>\n";

// Decision making using if-elseif-else
if ($hour >= 5 && $hour < 12) {
    $greeting = "Good Morning! 🌅";
} elseif ($hour >= 12 && $hour < 17) {
    $greeting = "Good Afternoon! ☀️";
} elseif ($hour >= 17 && $hour < 21) {
    $greeting = "Good Evening! 🌇";
} else {
    $greeting = "Good Night! 🌙";
}

echo "<h2>Greeting: $greeting</h2>";
?>
