// app/Http/Controllers/RailwayPortalController.php
namespace App\Http\Controllers;

use App\Models\RailwayBooking;
use App\Models\RailwaySystem;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class RailwayPortalController extends Controller
{
    // Requirement 4: Laravel Pagination with Division & User Filters
    public function bookings(Request $request)
    {
        $perPage = (int) $request->input('per_page', 15);
        $systemCode = $request->input('system');
        $userFilter = $request->input('user');
        $search = $request->input('search');

        $query = RailwayBooking::with(['user', 'railwaySystem'])->latest();

        // Filter by Railway Division (Requirement 6)
        if ($systemCode) {
            $query->where('system_code', $systemCode);
        }

        // Filter by User (Requirement 2 & 3: Minimum 100 entries per user)
        if ($userFilter) {
            $query->where(function($q) use ($userFilter) {
                $q->where('user_enrollment', $userFilter)
                  ->orWhere('passenger_name', 'like', "%{$userFilter}%");
            });
        }

        // Search query
        if ($search) {
            $query->where(function($q) use ($search) {
                $q->where('pnr', 'like', "%{$search}%")
                  ->orWhere('passenger_name', 'like', "%{$search}%")
                  ->orWhere('train_name', 'like', "%{$search}%");
            });
        }

        // Eloquent LengthAwarePaginator
        $bookings = $query->paginate($perPage)->withQueryString();
        $systems = RailwaySystem::with('adminUser')->get();
        $users = User::all();

        return view('railway.bookings', compact('bookings', 'systems', 'users'));
    }
}
