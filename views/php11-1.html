⚖️ Practical 11.1: Maximum and Minimum Numbers (Decision Making)

Problem Statement: Find out the maximum and minimum numbers from user entered values using conditional if-else structures.

PHP practical11_1.php
PHP 8.2 
<?php
/**
 * Practical 11.1: Maximum and Minimum Numbers
 * Group 10 — Government MCA College, Maninagar
 */

$numA = (float)($_POST['numA'] ?? 45);
$numB = (float)($_POST['numB'] ?? 18);
$numC = (float)($_POST['numC'] ?? 92);

// Decision Making: Maximum
if ($numA >= $numB && $numA >= $numC) {
    $max = $numA;
} elseif ($numB >= $numA && $numB >= $numC) {
    $max = $numB;
} else {
    $max = $numC;
}

// Decision Making: Minimum
if ($numA <= $numB && $numA <= $numC) {
    $min = $numA;
} elseif ($numB <= $numA && $numB <= $numC) {
    $min = $numB;
} else {
    $min = $numC;
}

echo "Numbers: $numA, $numB, $numC<br>";
echo "Maximum: " . $max . "<br>";
echo "Minimum: " . $min;
?>
