Problem Statement: Implement 5 specific string operations: (1) Print your name, (2) Print size of string passed as argument, (3) Concat two strings, (4) Convert case of string, (5) Find one string from another.

PHP practical11_4.php
PHP 8.2 
<?php
/**
 * Practical 11.4: PHP String Functions (5 Tasks)
 * Group 10 — Government MCA College, Maninagar
 */

// 11.4.1 Print your name
function printName($name) {
    return "Student Name: " . htmlspecialchars($name);
}

// 11.4.2 Size of string (passed as argument)
function getStringSize($str) {
    return strlen($str);
}

// 11.4.3 Concat two strings
function concatStrings($str1, $str2) {
    return $str1 . " " . $str2;
}

// 11.4.4 Convert case of string
function convertCases($str) {
    return [
        'upper' => strtoupper($str),
        'lower' => strtolower($str),
        'title' => ucwords(strtolower($str))
    ];
}

// 11.4.5 Find one string from another
function findSubstring($haystack, $needle) {
    $pos = strpos($haystack, $needle);
    return ($pos !== false)
        ? "Found at index: " . $pos
        : "Substring NOT found";
}
?>
