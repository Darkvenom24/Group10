Problem Statement: Demonstrate PHP array functions: (1) Print array values, (2) Reverse an array, (3) Merge two arrays in sorted manner, and (4) Add values of all elements of an array.

PHP practical10_1.php
PHP 8.2 
<?php
/**
 * Practical 10.1: Array Operations
 * Group 10 — Government MCA College, Maninagar
 */

$arr1_raw = $_POST['arr1'] ?? "14, 8, 35, 2, 19";
$arr2_raw = $_POST['arr2'] ?? "5, 42, 11, 23";

$arr1 = array_map('floatval', explode(",", $arr1_raw));
$arr2 = array_map('floatval', explode(",", $arr2_raw));

// 10.1.1 Print the values of array
echo "<h4>10.1.1 Print Values:</h4>";
foreach ($arr1 as $index => $value) {
    echo "Index [$index] => $value; ";
}

// 10.1.2 Reverse an array
echo "<h4>10.1.2 Reverse Array:</h4>";
$reversed = array_reverse($arr1);
print_r($reversed);

// 10.1.3 Merge two arrays in sorted manner
echo "<h4>10.1.3 Merge in Sorted Manner:</h4>";
$merged = array_merge($arr1, $arr2);
sort($merged);
print_r($merged);

// 10.1.4 Add values of all elements of an array
echo "<h4>10.1.4 Sum of Elements:</h4>";
$sum = array_sum($arr1);
echo "Total Sum: " . $sum;
?>
