Problem Statement: Connect a PHP script to a MySQL database using PDO, execute prepared SQL statements, handle errors with exceptions, and manage persistent relational records.

PHP practical12_crud.php
PHP 8.2 & PDO 
<?php
/**
 * Practical 12: MySQL Database Connection using PDO
 * Group 10 — Government MCA College, Maninagar
 */

$host = 'localhost';
$dbname = 'gmca_wtp_practical';
$username = 'root';
$password = '';

try {
    $dsn = "mysql:host=$host;dbname=$dbname;charset=utf8mb4";
    $options = [
        PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
        PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
        PDO::ATTR_EMULATE_PREPARES   => false,
    ];

    $pdo = new PDO($dsn, $username, $password, $options);
    echo "Database connection established successfully.<br>";

    // Prepared Statement Execution
    $stmt = $pdo->prepare("SELECT * FROM railway_bookings WHERE status = :status");
    $stmt->execute([':status' => 'Confirmed']);
    $rows = $stmt->fetchAll();

    echo "Total Confirmed Records: " . count($rows);

} catch (PDOException $e) {
    echo "Connection Failed: " . $e->getMessage();
}
?>
