<?php
/**
 * Practical 11.3: User Profile Web Page ($_POST Handling)
 * Group 10 — Government MCA College, Maninagar
 */

$isSubmitted = false;
$userData = [];

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $isSubmitted = true;
    $userData = [
        'fullName'   => htmlspecialchars(trim($_POST['fullName'] ?? '')),
        'enrollment' => htmlspecialchars(trim($_POST['enrollment'] ?? '')),
        'email'      => htmlspecialchars(trim($_POST['email'] ?? '')),
        'course'     => htmlspecialchars(trim($_POST['course'] ?? '')),
        'bio'        => htmlspecialchars(trim($_POST['bio'] ?? ''))
    ];
}

if ($isSubmitted) {
    echo "<h3>Profile Dossier for " . $userData['fullName'] . "</h3>";
    echo "Enrollment: " . $userData['enrollment'] . "<br>";
    echo "Email: " . $userData['email'] . "<br>";
    echo "Course: " . $userData['course'] . "<br>";
    echo "Biography: " . $userData['bio'];
}
?>
