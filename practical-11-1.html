Practical 11.1

Requirement: Write a PHP program to find out maximum and minimum number. Uses structured conditional statements (if-else) to compare the input numbers and identify both the maximum value and minimum value concurrently.

<?php
/**
 * Practical 11.1: Find out maximum and minimum number using if-else
 * Group 10 — Government MCA College, Maninagar (GMCA)
 */

if ($_SERVER["REQUEST_METHOD"] == "POST") {
    $a = (float)$_POST['valA'];
    $b = (float)$_POST['valB'];
    $c = (float)$_POST['valC'];

    // Decision Logic for Maximum
    if ($a >= $b && $a >= $c) {
        $max = $a;
    } elseif ($b >= $a && $b >= $c) {
        $max = $b;
    } else {
        $max = $c;
    }

    // Decision Logic for Minimum
    if ($a <= $b && $a <= $c) {
        $min = $a;
    } elseif ($b <= $a && $b <= $c) {
        $min = $b;
    } else {
        $min = $c;
    }

    echo "Given values: \$a=$a, \$b=$b, \$c=$c<br>\n";
    echo "Maximum Number: <strong>$max</strong><br>\n";
    echo "Minimum Number: <strong>$min</strong>\n";
}
?>
