⚡ Database Connection Code & CRUD Implementation

📄 config.php & practical12_crud.php (PDO Implementation) 
<?php
/**
 * Practical 12: Database Connection via PDO & CRUD Operations
 * Group 10 — Government MCA College, Maninagar (GMCA)
 */

$host = 'localhost';
$db   = 'gmca_group10_db';
$user = 'root';
$pass = '';
$charset = 'utf8mb4';

// 1. DATABASE CONNECTION VIA PDO
$dsn = "mysql:host=$host;dbname=$db;charset=$charset";
$options = [
    PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    PDO::ATTR_EMULATE_PREPARES   => false,
];

try {
    $pdo = new PDO($dsn, $user, $pass, $options);
    echo "Connected successfully to MySQL Database!";
} catch (PDOException $e) {
    die("Database Connection failed: " . $e->getMessage());
}

// 2. CREATE (Insert Practical 7 Form Data)
$stmt = $pdo->prepare("INSERT INTO practical7_students (enrollment, full_name, email, department, semester, project_title) 
                       VALUES (:enrollment, :full_name, :email, :department, :semester, :project_title)");
$stmt->execute([
    ':enrollment'    => '26GMCA52',
    ':full_name'     => 'Vaibhav Senjaliya',
    ':email'         => 'vaibhav@gmca.ac.in',
    ':department'    => 'MCA',
    ':semester'      => 'Semester 2',
    ':project_title' => 'Automated Railway Reservation Engine'
]);

// 3. READ (Retrieve records for Display Page)
$stmt = $pdo->query("SELECT * FROM practical7_students ORDER BY id DESC");
$records = $stmt->fetchAll();

// 4. UPDATE (Modify Student Record)
$updateStmt = $pdo->prepare("UPDATE practical7_students SET project_title = :title WHERE enrollment = :enroll");
$updateStmt->execute([':title' => 'Updated Project Name', ':enroll' => '26GMCA52']);

// 5. DELETE (Remove record by ID)
$delStmt = $pdo->prepare("DELETE FROM practical7_students WHERE id = :id");
$delStmt->execute([':id' => 1]);
?>
📄 config_mysqli.php (Procedural MySQLi Implementation) 
<?php
/**
 * Practical 12: Procedural MySQLi Connection & CRUD
 * Group 10 — Government MCA College, Maninagar (GMCA)
 */

$servername = "localhost";
$username   = "root";
$password   = "";
$dbname     = "gmca_group10_db";

// 1. Create Connection
$conn = mysqli_connect($servername, $username, $password, $dbname);

// Check Connection
if (!$conn) {
    die("Connection failed: " . mysqli_connect_error());
}
echo "Connected successfully using mysqli!";

// 2. CREATE (INSERT)
$sql = "INSERT INTO practical7_students (enrollment, full_name, email, department, semester, project_title) 
        VALUES ('26GMCA61', 'Ridham Bambhaniya', 'ridham@gmca.ac.in', 'MCA', 'Semester 2', 'Web Portal')";
mysqli_query($conn, $sql);

// 3. READ (SELECT)
$result = mysqli_query($conn, "SELECT * FROM practical7_students");
while ($row = mysqli_fetch_assoc($result)) {
    echo "Student: " . $row['full_name'] . " - " . $row['project_title'] . "<br>";
}

// 4. UPDATE
mysqli_query($conn, "UPDATE practical7_students SET semester='Semester 3' WHERE enrollment='26GMCA61'");

// 5. DELETE
mysqli_query($conn, "DELETE FROM practical7_students WHERE id=2");

// Close Connection
mysqli_close($conn);
?>

CREATE DATABASE IF NOT EXISTS `gmca_group10_db`;
USE `gmca_group10_db`;

-- Users Table for Index Page Login & Registration
CREATE TABLE IF NOT EXISTS `users` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `enrollment` VARCHAR(20) NOT NULL UNIQUE,
    `name` VARCHAR(100) NOT NULL,
    `email` VARCHAR(120) NOT NULL UNIQUE,
    `username` VARCHAR(50) NOT NULL UNIQUE,
    `password` VARCHAR(255) NOT NULL,
    `role` VARCHAR(30) DEFAULT 'Student',
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Practical 7 Students Table
CREATE TABLE IF NOT EXISTS `practical7_students` (
    `id` INT AUTO_INCREMENT PRIMARY KEY,
    `enrollment` VARCHAR(20) NOT NULL,
    `full_name` VARCHAR(100) NOT NULL,
    `email` VARCHAR(120) NOT NULL,
    `phone` VARCHAR(15) NOT NULL,
    `department` VARCHAR(100) NOT NULL,
    `semester` VARCHAR(20) NOT NULL,
    `skills` TEXT NOT NULL,
    `project_title` VARCHAR(255) NOT NULL,
    `project_desc` TEXT NULL,
    `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
